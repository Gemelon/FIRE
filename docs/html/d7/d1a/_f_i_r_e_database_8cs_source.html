708 public void Reload()
709 {
710 ThrowIfDisposed();
711
712 _records.Clear();
713 _records.AddRange(_context.FileRecords
714 .AsNoTracking()
715 .Include(record => record.FileMetaDatas)
716 .OrderBy(record => record.Id)
717 .Select(MapToModel));
718
719 // Migration: Compute FileIdHash for records that don't have it yet
720 var needsUpdate = false;
721 foreach (var record in _records)
722 {
723 if (record.FileId != null && string.IsNullOrEmpty(record.FileIdHash))
724 {
725 record.FileIdHash = ComputeFileIdHash(record.VolumeSerialNumber, record.FileId);
726 needsUpdate = true;
727 }
728 }
729
730 // If we computed any hashes, persist them back to the database
731 if (needsUpdate)
732 {
733 Save();
734 }
735
736 _fileIdIndex.Clear();
737 foreach (var record in _records)
738 {
739 if (record.FileId != null)
740 {
741 var key = CreateFileIdKey(record.VolumeSerialNumber, record.FileId);
742 _fileIdIndex.Add(key);
743 }
744 }
745
746 var statusEntity = _context.StatusRecords.AsNoTracking().SingleOrDefault(x => x.Id == FIREDatabaseStatusEntity.SingletonId);
747 if (statusEntity is not null)
748 {
749 CopyToModel(statusEntity, _statusRecord);
750 }
751
752 _counterStates.Clear();
753 foreach (var counterStateEntity in _context.CounterStates.AsNoTracking().OrderBy(x => x.ScopeKey))
754 {
755 _counterStates[counterStateEntity.ScopeKey] = new FIRECounterStateEntity
756 {
757 ScopeKey = counterStateEntity.ScopeKey,
758 LastValue = counterStateEntity.LastValue,
759 UpdatedAt = counterStateEntity.UpdatedAt
760 };
761 }

964 public bool FileExists(ulong volumeSerialNumber, byte[]? fileId)
965 {
966 if (fileId == null)
967 return false;
968
969 ThrowIfDisposed();
970
971 // First, try to find by hash (primary lookup)
972 var hash = ComputeFileIdHash(volumeSerialNumber, fileId);
973 if (hash != null && _records.Any(r => r.FileIdHash == hash))
974 return true;
975
976 // Fallback: search by composite VolumeSerialNumber + FileId (for backward compatibility with old records)
977 var key = CreateFileIdKey(volumeSerialNumber, fileId);
978 return _fileIdIndex.Contains(key);
979 }

987 internal FIREDbRecord? FindByFileIdentity(ulong volumeSerialNumber, byte[]? fileId)
988 {
989 if (fileId == null)
990 return null;
991
992 ThrowIfDisposed();
993
994 var hash = ComputeFileIdHash(volumeSerialNumber, fileId);
995 if (!string.IsNullOrWhiteSpace(hash))
996 {
997 var byHash = _records.FirstOrDefault(r => string.Equals(r.FileIdHash, hash, StringComparison.OrdinalIgnoreCase));
998 if (byHash is not null)
999 return byHash;
1000 }
1001
1002 return _records.FirstOrDefault(r =>
1003 r.VolumeSerialNumber == volumeSerialNumber &&
1004 r.FileId is not null &&
1005 r.FileId.SequenceEqual(fileId));
1006 }

1251 public static string? ComputeFileIdHash(ulong volumeSerialNumber, byte[]? fileId)
1252 {
1253 if (fileId == null)
1254 return null;
1255
1256 // Concatenate volume serial number (8 bytes) + fileId (16 bytes) = 24 bytes total
1257 byte[] combined = new byte[sizeof(ulong) + fileId.Length];
1258 BitConverter.GetBytes(volumeSerialNumber).CopyTo(combined, 0);
1259 fileId.CopyTo(combined, sizeof(ulong));
1260
1261 // Compute SHA256 hash
1262 using (var sha256 = SHA256.Create())
1263 {
1264 byte[] hashBytes = sha256.ComputeHash(combined);
1265 return Convert.ToHexString(hashBytes);
1266 }
1267 }

1442 {
1443 ThrowIfDisposed();
1444 ArgumentException.ThrowIfNullOrWhiteSpace(scopeKey);

1446 scopeKey = scopeKey.Trim();

1453 LastValue = 0,
1454 UpdatedAt = DateTime.UtcNow

1500 internal void ResetCounterStatesToExecuted()
1501 {
1502 ThrowIfDisposed();
1503
1504 // Build new counter states from executed records only.
1505 var executedMaxCounters = new Dictionary<string, long>(StringComparer.OrdinalIgnoreCase);
1506
1507 foreach (var record in _records)
1508 {
1509 if (record.Status != ProcessingStatus.Executed) continue;
1510 if (record.AssignedCounterValue is not { } counterValue) continue;
1511
1512 var scope = Path.GetDirectoryName(record.TargetFilePath ?? string.Empty) ?? string.Empty;
1513 if (string.IsNullOrWhiteSpace(scope)) continue;
1514
1515 if (!executedMaxCounters.TryGetValue(scope, out var current) || counterValue > current)
1516 executedMaxCounters[scope] = counterValue;
1517 }
1518
1519 // Replace in-memory counter states with the computed maxima.
1520 _counterStates.Clear();
1521 foreach (var (scope, maxValue) in executedMaxCounters)
1522 {
1523 _counterStates[scope] = new FIRECounterStateEntity
1524 {
1525 ScopeKey = scope,
1526 LastValue = maxValue,
1527 UpdatedAt = DateTime.UtcNow
1528 };
1529 }
1530
1531 // Clear the pending counter assignments so they will be re-assigned cleanly.
1532 foreach (var record in _records)
1533 {
1534 if (record.Status != ProcessingStatus.Executed)
1535 record.AssignedCounterValue = null;
1536 }
1537 }

1561 private void PersistCurrentState(string status, bool valid, string? errorMessage)
1562 {
1563 try
1564 {
1565 using var transaction = _context.Database.BeginTransaction();
1566
1567 var existingRecords = _context.FileRecords.Include(record => record.FileMetaDatas).ToList();
1568 if (existingRecords.Count > 0)
1569 {
1570 _context.FileMetaDataRecords.RemoveRange(existingRecords.SelectMany(record => record.FileMetaDatas));
1571 _context.FileRecords.RemoveRange(existingRecords);
1572 _context.SaveChanges();
1573 }
1574
1575 var existingCounterStates = _context.CounterStates.ToList();
1576 if (existingCounterStates.Count > 0)
1577 {
1578 _context.CounterStates.RemoveRange(existingCounterStates);
1579 _context.SaveChanges();
1580 }
1581
1582 foreach (var record in _records)
1583 {
1584 _context.FileRecords.Add(MapToEntity(record));
1585 }
1586
1587 foreach (var counterState in _counterStates.Values)
1588 {
1589 _context.CounterStates.Add(new FIRECounterStateEntity
1590 {
1591 ScopeKey = counterState.ScopeKey,
1592 LastValue = counterState.LastValue,
1593 UpdatedAt = counterState.UpdatedAt
1594 });
1595 }
1596
1597 var statusEntity = _context.StatusRecords.Single(x => x.Id == FIREDatabaseStatusEntity.SingletonId);
1598 statusEntity.Status = status;
1599 statusEntity.Valid = valid;
1600 statusEntity.ErrorMessage = errorMessage;
1601 statusEntity.TimeStamp = DateTime.UtcNow;
1602
1603 _context.SaveChanges();
1604 transaction.Commit();
1605 CopyToModel(statusEntity, _statusRecord);
1606 }
1607 catch (Exception ex)
1608 {
1609 try
1610 {
1611 var statusEntity = _context.StatusRecords.Single(x => x.Id == FIREDatabaseStatusEntity.SingletonId);
1612 statusEntity.Status = "Failed";
1613 statusEntity.Valid = false;
1614 statusEntity.ErrorMessage = ex.Message;
1615 statusEntity.TimeStamp = DateTime.UtcNow;
1616 _context.SaveChanges();
1617 CopyToModel(statusEntity, _statusRecord);
1618 }
1619 catch
1620 {
1621 // If the status cannot be updated, the original exception is preserved.
1622 }
1623
1624 throw;
1625 }

1759 private void ThrowIfDisposed()
1760 {

1787 internal void SortRecords(List<string>? fileSorting, string? fileSortingOrder)
1788 {
1789 ThrowIfDisposed();
1790
1791 // Nothing to sort when there are zero/one records or no sorting definition.
1792 if (_records.Count <= 1 || fileSorting is null || fileSorting.Count == 0)
1793 return;
1794
1795 // Resolve global default direction first (used when no field-specific direction is provided).
1796 var globalDirection = ParseGlobalSortDirection(fileSortingOrder);
1797 var criteria = new List<SortCriterion>();
1798
1799 // Parse every configured criterion and preserve the configured order.
1800 foreach (var rawCriterion in fileSorting)
1801 {
1802 if (TryParseSortCriterion(rawCriterion, globalDirection, out var criterion))
1803 {
1804 criteria.Add(criterion);
1805 }
1806 }
1807
1808 // Abort if no valid criterion remained after parsing.
1809 if (criteria.Count == 0)
1810 return;
1811
1812 IOrderedEnumerable<FIREDbRecord>? ordered = null;
1813
1814 // Build a dynamic multi-column sort chain: first criterion => OrderBy, subsequent => ThenBy.
1815 foreach (var criterion in criteria)
1816 {
1817 if (ordered is null)
1818 {
1819 ordered = criterion.Direction == SortDirection.Descending
1820 ? _records.OrderByDescending(record => GetSortValue(record, criterion.Name), SortValueComparer.Instance)
1821 : _records.OrderBy(record => GetSortValue(record, criterion.Name), SortValueComparer.Instance);
1822 }
1823 else
1824 {
1825 ordered = criterion.Direction == SortDirection.Descending
1826 ? ordered.ThenByDescending(record => GetSortValue(record, criterion.Name), SortValueComparer.Instance)
1827 : ordered.ThenBy(record => GetSortValue(record, criterion.Name), SortValueComparer.Instance);
1828 }
1829 }
1830
1831 if (ordered is null)
1832 return;
1833
1834 // Deterministic tie-breakers avoid unstable processing order for identical sort values.
1835 var sorted = ordered
1836 .ThenBy(record => record.SourceFilePath ?? string.Empty, StringComparer.OrdinalIgnoreCase)
1837 .ThenBy(record => record.TargetFilePath ?? string.Empty, StringComparer.OrdinalIgnoreCase)
1838 .ToList();
1839
1840 // Replace the in-memory sequence so generate/execute consume the new order.
1841 _records.Clear();
1842 _records.AddRange(sorted);
1843 }

1872 private static bool TryParseSortCriterion(string? rawCriterion, SortDirection defaultDirection, out SortCriterion criterion)
1873 {
1874 criterion = default;
1875
1876 if (string.IsNullOrWhiteSpace(rawCriterion))
1877 return false;
1878
1879 var token = rawCriterion.Trim();
1880
1881 // Per-field direction is encoded as <criterion>:<direction>.
1882 var separatorIndex = token.LastIndexOf(':');
1883 if (separatorIndex > 0 && separatorIndex < token.Length - 1)
1884 {
1885 var criterionName = token[..separatorIndex].Trim();
1886 var directionToken = token[(separatorIndex + 1)..].Trim();
1887
1888 if (!string.IsNullOrWhiteSpace(criterionName) && TryParseSortDirection(directionToken, out var parsedDirection))
1889 {
1890 criterion = new SortCriterion(criterionName, parsedDirection);
1891 return true;
1892 }
1893 }
1894
1895 // No valid field-specific direction found => fall back to global direction.
1896 criterion = new SortCriterion(token, defaultDirection);
1897 return true;
1898 }

1906 private static bool TryParseSortDirection(string? value, out SortDirection direction)
1907 {
1908 direction = SortDirection.Ascending;
1909
1910 if (string.IsNullOrWhiteSpace(value))
1911 return false;
1912
1913 switch (value.Trim().ToUpperInvariant())
1914 {
1915 case "ASC":
1916 case "ASCENDING":
1917 case "AUFSTEIGEND":
1918 direction = SortDirection.Ascending;
1919 return true;
1920 case "DESC":
1921 case "DESCENDING":
1922 case "ABSTEIGEND":
1923 direction = SortDirection.Descending;
1924 return true;
1925 default:
1926 return false;
1927 }

1965 {

1967 return 0;
1968

1970 if (x is null)
1971 return 1;
1972 if (y is null)
1973 return -1;

1979 var right = Convert.ToString(y, CultureInfo.InvariantCulture) ?? string.Empty;

2008 private static IComparable? GetSortValue(FIREDbRecord record, string criterion)
2009 {
2010 var parts = criterion.Split('.', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries);
2011 if (parts.Length == 0)
2012 return null;
2013
2014 var key = parts[0];
2015 var selector = parts.Length > 1 ? parts[1] : null;
2016
2017 var metadataValue = record.FileMetaDatas.FirstOrDefault(m =>
2018 string.Equals(m.Key, key, StringComparison.OrdinalIgnoreCase))?.Value;
2019
2020 if (string.IsNullOrWhiteSpace(metadataValue))
2021 return null;
2022
2023 if (selector == null)
2024 {
2025 if (DateTime.TryParse(metadataValue, CultureInfo.InvariantCulture, DateTimeStyles.AllowWhiteSpaces, out var dateTimeValue))
2026 return dateTimeValue;
2027
2028 if (decimal.TryParse(metadataValue, NumberStyles.Any, CultureInfo.InvariantCulture, out var numberValue))
2029 return numberValue;
2030
2031 return metadataValue;
2032 }
2033
2034 if (!DateTime.TryParse(metadataValue, CultureInfo.InvariantCulture, DateTimeStyles.AllowWhiteSpaces, out var parsedDateTime))
2035 return metadataValue;
2036
2037 return selector.ToLowerInvariant() switch
2038 {
2039 "year" => parsedDateTime.Year,
2040 "month" => parsedDateTime.Month,
2041 "day" => parsedDateTime.Day,
2042 "hour" => parsedDateTime.Hour,
2043 "minute" => parsedDateTime.Minute,
2044 "second" => parsedDateTime.Second,
2045 _ => metadataValue
2046 };
2047 }
