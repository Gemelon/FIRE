151public sealed class FileInfoMetadataSource : IMetadataSource

158
167 public Dictionary<string, string> ExtractMetadata(string filePath)
168 {
169 var metadata = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
170 try
171 {
172 var fileInfo = new FileInfo(filePath);
173 if (!fileInfo.Exists) return metadata;
174
175 metadata["CreationTime"] = fileInfo.CreationTime.ToString("yyyy-MM-dd HH:mm:ss", CultureInfo.InvariantCulture);
176 metadata["ModificationTime"] = fileInfo.LastWriteTime.ToString("yyyy-MM-dd HH:mm:ss", CultureInfo.InvariantCulture);
177 metadata["AccessTime"] = fileInfo.LastAccessTime.ToString("yyyy-MM-dd HH:mm:ss", CultureInfo.InvariantCulture);
178 }
179 catch { }
180 return metadata;
181 }
182}

277 public Dictionary<string, string> ExtractMetadata(string filePath)
278 {
279 var metadata = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
280
281 try
282 {
283 if (!File.Exists(filePath))
284 {
285 return metadata;
286 }
287
288 // Execute ExifTool synchronously and get all metadata
289 var result = _exifTool.ExtractAllMetadataAsync(filePath).GetAwaiter().GetResult();
290
291 if (result != null)
292 {
293 foreach (var kvp in result)
294 {
295 if (!string.IsNullOrWhiteSpace(kvp.Key) && kvp.Value != null)
296 {
297 metadata[kvp.Key] = kvp.Value.ToString() ?? string.Empty;
298 }
299 }
300 }
301 }
302 catch
303 {
304 // Metadata extraction failed; return partial or empty result.
305 }
306
307 return metadata;
308 }

318 {
319 if (_disposed)
320 {
321 return;
322 }
323
324 _exifTool?.Dispose();

853 public void GenerateTargetPaths(Action<string>? progressCallback = null)
854 {
855 ThrowIfDisposed();
856
857 _database.SortRecords(_configuration.FileSorting, _configuration.FileSortingOrder);
858
859 // Reset per-scope counter states to the maximum value already committed by Executed
860 // records. This ensures that repeated Generate invocations before Execute do not
861 // keep incrementing the counter beyond what was last assigned.
862 _database.ResetCounterStatesToExecuted();
863
864 var pendingCount = _database.Count(record =>
865 !string.IsNullOrWhiteSpace(record.SourceFilePath) &&
866 record.Status is not ProcessingStatus.Executed &&
867 !record.ExcludedFromGenerate);
868
869 BeginStage(FIRECatalogStage.Generate, pendingCount);
870
871 foreach (var record in _database)
872 {
873 if (string.IsNullOrWhiteSpace(record.SourceFilePath)) continue;
874
875 // Skip files that have already been executed – their paths are immutable.
876 if (record.Status is ProcessingStatus.Executed) continue;
877
878 // Skip files explicitly excluded from Generate phase
879 if (record.ExcludedFromGenerate)
880 {
881 _logger?.Log(FIRELogLevel.Info, "Generate", $"Skipped (excluded by user): {record.SourceFilePath}");
882 continue;
883 }
884
885 var progressPath = record.SourceFilePath;
886
887 // Handle sidecar files separately
888 if (record.Classification == FileClassification.SidecarFile)
889 {
890 GenerateSidecarTargetPath(record);
891 // Update status for sidecars
892 if (!string.IsNullOrWhiteSpace(record.TargetFilePath))
893 record.Status = ProcessingStatus.PathGenerated;
894
895 progressPath = record.TargetFilePath ?? progressPath;
896 progressCallback?.Invoke(progressPath);
897 ReportFileProgress(progressPath);
898 continue;
899 }
900
901 var extension = Path.GetExtension(record.SourceFilePath).ToLowerInvariant();
902 if (!_configuration.FileExtensions.TryGetValue(extension, out var extConfig))
903 {
904 progressCallback?.Invoke(progressPath);
905 ReportFileProgress(progressPath);
906 continue;
907 }
908
909 ApplyMetadataRules(record, extConfig);
910
911 var metadataLookup = record.FileMetaDatas.ToDictionary(
912 m => m.Key ?? string.Empty,
913 m => m.Value ?? string.Empty,
914 StringComparer.OrdinalIgnoreCase);
915 var metadataTypeLookup = record.FileMetaDatas.ToDictionary(
916 m => m.Key ?? string.Empty,
917 m => m.TypeName ?? string.Empty,
918 StringComparer.OrdinalIgnoreCase);
919
920 var sortingPattern = extConfig.SortingPatern ?? _configuration.SortingPatern ?? string.Empty;
921 var fileNamePattern = extConfig.FileNamePatern ?? _configuration.FileNamePatern ?? string.Empty;
922
923 var targetDirectory = ParseTemplate(sortingPattern, metadataLookup, record.SourceFilePath, null, metadataTypeLookup);
924 if (string.IsNullOrWhiteSpace(targetDirectory))
925 {
926 progressCallback?.Invoke(progressPath);
927 ReportFileProgress(progressPath);
928 continue;
929 }
930
931 long? counter = null;
932 if (ContainsCounterPlaceholder(fileNamePattern))
933 {
934 counter = _database.GetNextCounterValue(targetDirectory);
935 record.AssignedCounterValue = counter;
936 }
937 else
938 {
939 record.AssignedCounterValue = null;
940 }
941
942 var targetFileName = ParseTemplate(fileNamePattern, metadataLookup, record.SourceFilePath, counter, metadataTypeLookup);
943
944 if (!string.IsNullOrWhiteSpace(targetFileName))
945 {
946 record.TargetFilePath = Path.Combine(targetDirectory, targetFileName);
947 // Update status after successful path generation
948 record.Status = ProcessingStatus.PathGenerated;
949 _logger?.LogContinuation(FIRELogLevel.Debug, record.TargetFilePath);
950 }
951
952 progressPath = record.TargetFilePath ?? progressPath;
953 progressCallback?.Invoke(progressPath);
954 ReportFileProgress(progressPath);
955 }
956
957 _database.Save();
958 CompleteCurrentStage();
959 }

980 private void GenerateSidecarTargetPath(FIREDbRecord sidecarRecord)
981 {
982 if (string.IsNullOrWhiteSpace(sidecarRecord.SourceFilePath))
983 return;
984
985 var sidecarDirectory = Path.GetDirectoryName(sidecarRecord.SourceFilePath);
986 var sidecarBaseName = Path.GetFileNameWithoutExtension(sidecarRecord.SourceFilePath);
987 var sidecarExtension = Path.GetExtension(sidecarRecord.SourceFilePath);
988
989 if (string.IsNullOrWhiteSpace(sidecarDirectory) || string.IsNullOrWhiteSpace(sidecarBaseName))
990 return;
991
992 // Find the primary file in the same directory with the same base name
993 var primaryRecord = _database.FirstOrDefault(r =>
994 r.Classification == FileClassification.RegularFile &&
995 !string.IsNullOrWhiteSpace(r.SourceFilePath) &&
996 string.Equals(Path.GetDirectoryName(r.SourceFilePath), sidecarDirectory, StringComparison.OrdinalIgnoreCase) &&
997 string.Equals(Path.GetFileNameWithoutExtension(r.SourceFilePath), sidecarBaseName, StringComparison.OrdinalIgnoreCase));
998
999 if (primaryRecord == null || string.IsNullOrWhiteSpace(primaryRecord.TargetFilePath))
1000 {
1001 LogMetadataWarning(sidecarRecord.SourceFilePath, "Sidecar", "could not find primary file or primary file has no target path");
1002 return;
1003 }
1004
1005 // Inherit target directory and base name from primary, keep sidecar extension
1006 var primaryTargetDirectory = Path.GetDirectoryName(primaryRecord.TargetFilePath);
1007 var primaryTargetBaseName = Path.GetFileNameWithoutExtension(primaryRecord.TargetFilePath);
1008
1009 if (!string.IsNullOrWhiteSpace(primaryTargetDirectory) && !string.IsNullOrWhiteSpace(primaryTargetBaseName))
1010 {
1011 sidecarRecord.TargetFilePath = Path.Combine(primaryTargetDirectory, primaryTargetBaseName + sidecarExtension);
1012 }
1013 }

1040 public void ExecuteFileOperations(Action<string>? progressCallback = null)
1041 {
1042 ThrowIfDisposed();
1043
1044 var pendingCount = _database.Count(record =>
1045 !string.IsNullOrWhiteSpace(record.SourceFilePath) &&
1046 !string.IsNullOrWhiteSpace(record.TargetFilePath) &&
1047 record.Status != ProcessingStatus.Executed &&
1048 !record.ExcludedFromExecute &&
1049 File.Exists(record.SourceFilePath));
1050
1051 BeginStage(FIRECatalogStage.Execute, pendingCount);
1052
1053 foreach (var record in _database)
1054 {
1055 if (string.IsNullOrWhiteSpace(record.SourceFilePath) || string.IsNullOrWhiteSpace(record.TargetFilePath)) continue;
1056 if (!File.Exists(record.SourceFilePath)) continue;
1057
1058 // Skip files that have already been executed (incremental workflow)
1059 if (record.Status == ProcessingStatus.Executed) continue;
1060
1061 // Skip files explicitly excluded from Execute phase
1062 if (record.ExcludedFromExecute)
1063 {
1064 _logger?.Log(FIRELogLevel.Info, "Execute", $"Skipped (excluded by user): {record.SourceFilePath}");
1065 continue;
1066 }
1067
1068 // Skip files that don't have a generated target path (e.g., excluded from Generate)
1069 if (record.Status == ProcessingStatus.NotProcessed)
1070 {
1071 _logger?.Log(FIRELogLevel.Info, "Execute", $"Skipped (not generated): {record.SourceFilePath}");
1072 continue;
1073 }
1074
1075 progressCallback?.Invoke(record.TargetFilePath);
1076 ReportFileProgress(record.TargetFilePath);
1077
1078 string action;
1079
1080 // Sidecar files inherit the action from their primary file
1081 if (record.Classification == FileClassification.SidecarFile)
1082 {
1083 // Use global default action for sidecars
1084 action = _configuration.Action ?? "Copy";
1085 }
1086 else
1087 {
1088 // Regular files use their extension-specific configuration
1089 var extension = Path.GetExtension(record.SourceFilePath).ToLowerInvariant();
1090 if (!_configuration.FileExtensions.TryGetValue(extension, out var extConfig)) continue;
1091 action = extConfig.Action ?? _configuration.Action ?? "Copy";
1092 }
1093
1094 ExecuteAction(action, record.SourceFilePath, record.TargetFilePath);
1095
1096 // Mark as executed after successful operation
1097 record.Status = ProcessingStatus.Executed;
1098 }
1099
1100 _database.Save();
1101 CompleteCurrentStage();
1102 }
