320 public static FIREConfigration Parse(string yaml)
321 {
322 ArgumentException.ThrowIfNullOrWhiteSpace(yaml);
323
324 var deserializer = new DeserializerBuilder()
325 .WithNamingConvention(NullNamingConvention.Instance)
326 .IgnoreUnmatchedProperties()
327 .Build();
328
329 var configuration = deserializer.Deserialize<FIREConfigration>(yaml) ?? new FIREConfigration();
330 configuration.Normalize();
331 return configuration;
332 }

370 private void Normalize()
371 {
372 DataBasePath ??= string.Empty;
373 DataBaseFileName ??= string.Empty;
374 Action = string.IsNullOrWhiteSpace(Action) ? "Copy" : Action;
375 MediaRootPath = string.IsNullOrWhiteSpace(MediaRootPath) ? "CC" : MediaRootPath;
376 RootPath = string.IsNullOrWhiteSpace(RootPath) ? "{MediaRootPath}" : RootPath;
377 SortingPatern ??= string.Empty;
378 FileNamePatern ??= string.Empty;
379 FileSorting ??= [];
380 FileSortingOrder = string.IsNullOrWhiteSpace(FileSortingOrder) ? "Ascending" : FileSortingOrder;
381 StringReplacements ??= new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
382 MetadataRules ??= [];
383 FileExtensions ??= new Dictionary<string, FileExtensionConfiguration>(StringComparer.OrdinalIgnoreCase);
384
385 foreach (var rule in MetadataRules)
386 {
387 rule.Normalize();
388 }
389
390 foreach (var item in FileExtensions.Values)
391 {
392 item.Normalize();
393 }
394
395 Logging?.Normalize(DataBasePath);
396 }

861 LogLevel = string.IsNullOrWhiteSpace(LogLevel) ? "Error" : LogLevel;
862 MaxFileSizeBytes = MaxFileSizeBytes <= 0 ? 10 * 1024 * 1024 : MaxFileSizeBytes;
863 MaxAgeDays = MaxAgeDays < 0 ? 30 : MaxAgeDays;
864
865 if (string.IsNullOrWhiteSpace(LogFilePath))
866 {
867 LogFilePath = string.IsNullOrWhiteSpace(databasePath)
868 ? Path.Combine(AppContext.BaseDirectory, "Logs")
869 : Path.Combine(databasePath, "Logs");
870 }
